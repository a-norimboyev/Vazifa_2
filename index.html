    users {
        bigint id PK
        varchar username UK
        varchar email UK
        varchar password_hash
        varchar full_name
        text avatar_url
        varchar status
    }

    workspaces {
        bigint id PK
        varchar name
        varchar slug UK
        varchar visibility
        bigint owner_id FK
    }

    workspace_members {
        bigint id PK
        bigint workspace_id FK
        bigint user_id FK
        varchar role
    }

    template_categories {
        bigint id PK
        varchar name
        varchar slug UK
        varchar icon
    }

    boards {
        bigint id PK
        bigint workspace_id FK
        varchar title
        varchar visibility
        boolean is_template
        bigint template_category_id FK
        bigint created_by FK
    }

    board_members {
        bigint id PK
        bigint board_id FK
        bigint user_id FK
        varchar role
    }

    lists {
        bigint id PK
        bigint board_id FK
        varchar title
        float position
        boolean is_archived
    }

    cards {
        bigint id PK
        bigint list_id FK
        bigint board_id FK
        varchar title
        text description
        float position
        timestamp due_date
        boolean is_completed
        boolean is_template
        bigint created_by FK
    }

    card_members {
        bigint id PK
        bigint card_id FK
        bigint user_id FK
    }

    labels {
        bigint id PK
        bigint board_id FK
        varchar name
        varchar color
    }

    card_labels {
        bigint id PK
        bigint card_id FK
        bigint label_id FK
    }

    checklists {
        bigint id PK
        bigint card_id FK
        varchar title
        float position
    }

    checklist_items {
        bigint id PK
        bigint checklist_id FK
        text content
        boolean is_checked
        float position
        bigint assigned_to FK
    }

    attachments {
        bigint id PK
        bigint card_id FK
        bigint uploader_id FK
        varchar file_name
        text file_url
        bigint file_size
    }

    comments {
        bigint id PK
        bigint card_id FK
        bigint user_id FK
        text content
    }

    activities {
        bigint id PK
        bigint board_id FK
        bigint card_id FK
        bigint user_id FK
        varchar action_type
        jsonb action_data
    }

    notifications {
        bigint id PK
        bigint recipient_id FK
        bigint sender_id FK
        bigint activity_id FK
        text content
        boolean is_read
    }

🚀 Onlayn Havola (Link) Olish Bo‘yicha Qo‘llanma

Tekshiruvchiga topshirish uchun onlayn havola olish juda oson:

  1. dbdiagram.io orqali: dbdiagram.io saytiga kiring, ushbu papkadagi trello_schema.dbml fayli ichidagi barcha matnni nusxalab saytdagi maydonga joylang. So‘ng "Share" tugmasi orqali ommaviy havola linkini oling.
  2. drawsql.app orqali: drawsql.app saytiga kiring, yangi PostgreSQL diagrammasi ochib, "Import SQL" bo‘limiga papkadagi trello_schema.sql faylini joylang va havolani nusxalang.
